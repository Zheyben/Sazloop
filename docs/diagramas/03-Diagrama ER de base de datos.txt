Project MealFlow {
  database_type: 'MySQL'
  Note: 'Modelo ER derivado de las 31 entidades JPA entregadas por el equipo.'
}

// =====================================================
// MODULO: IDENTIDAD
// =====================================================

Table usuarios {
  id bigint [pk, increment]
  nombres varchar(80) [not null]
  apellidos varchar(80) [not null]
  email varchar(150) [not null, unique]
  password_hash varchar(255) [not null]
  telefono varchar(20)
  activo boolean [not null]
  fecha_registro datetime [not null]
}

Table consumidores {
  id bigint [pk, increment]
  usuario_id bigint [not null, unique]
  activo boolean [not null]
  fecha_creacion datetime [not null]
}

Table propietarios_empresa {
  id bigint [pk, increment]
  usuario_id bigint [not null, unique]
  activo boolean [not null]
  fecha_creacion datetime [not null]
}

Table empresas {
  id bigint [pk, increment]
  razon_social varchar(150) [not null]
  nombre_comercial varchar(120) [not null]
  ruc varchar(11) [not null, unique]
  propietario_id bigint [not null, unique]
  activa boolean [not null]
  fecha_registro datetime [not null]
}

Table direcciones {
  id bigint [pk, increment]
  departamento varchar(80) [not null]
  provincia varchar(80) [not null]
  distrito varchar(80) [not null]
  direccion_linea varchar(180) [not null]
  numero varchar(20)
  referencia varchar(250)
  codigo_postal varchar(10)
}

Table establecimientos {
  id bigint [pk, increment]
  empresa_id bigint [not null]
  nombre varchar(120) [not null]
  descripcion varchar(500)
  telefono varchar(20)
  direccion_id bigint [not null, unique]
  estado varchar(30) [not null, note: 'PENDIENTE_APROBACION, ACTIVO, SUSPENDIDO o INACTIVO']
  fecha_registro datetime [not null]
}

Table cuentas_establecimiento {
  id bigint [pk, increment]
  establecimiento_id bigint [not null, unique]
  nombre_usuario varchar(80) [not null, unique]
  password_hash varchar(255) [not null]
  activa boolean [not null]
  fecha_creacion datetime [not null]
  ultimo_acceso datetime
}

// =====================================================
// MODULO: AFILIACION
// =====================================================

Table administradores_plataforma {
  id bigint [pk, increment]
  nombre varchar(120) [not null]
  email_institucional varchar(150) [not null, unique]
  password_hash varchar(255) [not null]
  nivel_acceso varchar(20) [not null, note: 'ADMINISTRADOR o SUPER_ADMINISTRADOR']
  activo boolean [not null]
  fecha_creacion datetime [not null]
}

Table solicitudes_afiliacion {
  id bigint [pk, increment]
  establecimiento_id bigint [not null]
  fecha_solicitud datetime [not null]
  estado varchar(20) [not null, note: 'PENDIENTE, EN_REVISION, APROBADA o RECHAZADA']
}

Table documentos_establecimiento {
  id bigint [pk, increment]
  tipo_documento varchar(40) [not null, note: 'LICENCIA_FUNCIONAMIENTO, CERTIFICADO_SANITARIO, INSPECCION_SEGURIDAD u OTRO']
  numero_documento varchar(80)
  url_archivo varchar(500) [not null]
  fecha_emision date
  fecha_vencimiento date
  obligatorio boolean [not null]
  fecha_carga datetime [not null]
  solicitud_id bigint [not null]
}

Table revisiones_documento {
  id bigint [pk, increment]
  documento_id bigint [not null]
  administrador_id bigint [not null]
  resultado varchar(20) [not null, note: 'APROBADO o RECHAZADO']
  observaciones varchar(500)
  fecha_revision datetime [not null]
}

Table evaluaciones_afiliacion {
  id bigint [pk, increment]
  solicitud_id bigint [not null, unique]
  administrador_id bigint [not null]
  decision varchar(20) [not null, note: 'APROBADA o RECHAZADA']
  motivo varchar(500)
  fecha_resolucion datetime [not null]
}

Table suspensiones_establecimiento {
  id bigint [pk, increment]
  establecimiento_id bigint [not null]
  administrador_id bigint [not null]
  motivo varchar(500) [not null]
  fecha_inicio datetime [not null]
  fecha_fin datetime
  activa boolean [not null]
}

// =====================================================
// MODULO: OFERTAS
// =====================================================

Table packs_sorpresa {
  id bigint [pk, increment]
  establecimiento_id bigint [not null]
  nombre varchar(120) [not null]
  descripcion varchar(500) [not null]
  posibles_productos varchar(1000) [not null]
  activo boolean [not null]
  fecha_creacion datetime [not null]
}

Table alergenos {
  id bigint [pk, increment]
  nombre varchar(80) [not null, unique]
  descripcion varchar(300)
  activo boolean [not null]
}

Table restricciones_alimentarias {
  id bigint [pk, increment]
  nombre varchar(100) [not null, unique]
  descripcion varchar(300)
  activa boolean [not null]
}

Table pack_alergenos {
  pack_id bigint [not null]
  alergeno_id bigint [not null]

  indexes {
    (pack_id, alergeno_id) [pk]
  }
}

Table pack_restricciones {
  pack_id bigint [not null]
  restriccion_id bigint [not null]

  indexes {
    (pack_id, restriccion_id) [pk]
  }
}

Table ofertas_pack {
  id bigint [pk, increment]
  pack_id bigint [not null]
  fecha_oferta date [not null]
  inicio_recojo datetime [not null]
  fin_recojo datetime [not null]
  limite_definitivo datetime [not null]
  estado varchar(20) [not null, note: 'BORRADOR, ACTIVA, AGOTADA, FINALIZADA o CANCELADA']
  fecha_publicacion datetime [not null]
}

Table variantes_pack {
  id bigint [pk, increment]
  oferta_id bigint [not null]
  tamanio varchar(20) [not null, note: 'PEQUENO, MEDIANO o GRANDE']
  precio_referencial decimal(10,2) [not null]
  precio_descuento decimal(10,2) [not null]
  stock_inicial int [not null]
  stock_disponible int [not null]
  activa boolean [not null]
}

// =====================================================
// MODULO: RESERVAS
// =====================================================

Table retenciones_stock {
  id bigint [pk, increment]
  variante_id bigint [not null]
  consumidor_id bigint [not null]
  cantidad int [not null]
  fecha_creacion datetime [not null]
  fecha_expiracion datetime [not null]
  estado varchar(20) [not null, note: 'ACTIVA, CONFIRMADA, LIBERADA o VENCIDA']
  motivo varchar(25) [not null, note: 'COMPRA_DIGITAL o EFECTIVO_PRESENCIAL']
}

Table reservas {
  id bigint [pk, increment]
  consumidor_id bigint [not null]
  oferta_id bigint [not null]
  monto_total decimal(10,2) [not null]
  estado varchar(20) [not null, note: 'PENDIENTE_PAGO, CONFIRMADA, RECOGIDA, CANCELADA o EXPIRADA']
  origen_compra varchar(20) [not null, note: 'REMOTA o PRESENCIAL_QR']
  fecha_creacion datetime [not null]
  fecha_confirmacion datetime
}

Table detalles_reserva {
  id bigint [pk, increment]
  reserva_id bigint [not null]
  variante_id bigint [not null]
  cantidad int [not null]
  precio_unitario decimal(10,2) [not null]
  subtotal decimal(10,2) [not null]
}

Table codigos_recojo {
  id bigint [pk, increment]
  reserva_id bigint [not null, unique]
  codigo varchar(20) [not null, unique]
  estado varchar(15) [not null, note: 'PENDIENTE, UTILIZADO o VENCIDO']
  fecha_generacion datetime [not null]
  fecha_uso datetime
}

Table recojos {
  id bigint [pk, increment]
  reserva_id bigint [not null, unique]
  cuenta_establecimiento_id bigint [not null]
  fecha_recojo datetime [not null]
  observaciones varchar(300)
}

// =====================================================
// MODULO: TRANSACCIONES
// =====================================================

Table pagos {
  id bigint [pk, increment]
  reserva_id bigint [not null, unique]
  importe decimal(10,2) [not null]
  medio_pago varchar(20) [not null, note: 'TARJETA_DEBITO, YAPE o EFECTIVO']
  contexto varchar(20) [not null, note: 'REMOTO o PRESENCIAL_QR']
  estado varchar(20) [not null, note: 'PENDIENTE, CONFIRMADO o RECHAZADO']
  simulado boolean [not null]
  fecha_creacion datetime [not null]
  fecha_confirmacion datetime
}

Table boletas_simuladas {
  id bigint [pk, increment]
  reserva_id bigint [not null, unique]
  numero varchar(30) [not null, unique]
  total decimal(10,2) [not null]
  fecha_emision datetime [not null]
  mensaje_legal varchar(150) [not null]
}

Table cancelaciones_reserva {
  id bigint [pk, increment]
  reserva_id bigint [not null, unique]
  motivo varchar(500)
  importe_reembolso decimal(10,2) [not null]
  importe_penalizacion decimal(10,2) [not null]
  stock_devuelto boolean [not null]
  fecha_cancelacion datetime [not null]
}

Table reembolsos {
  id bigint [pk, increment]
  cancelacion_id bigint [not null, unique]
  importe decimal(10,2) [not null]
  estado varchar(20) [not null, note: 'PENDIENTE, PROCESADO o RECHAZADO']
  simulado boolean [not null]
  fecha_registro datetime [not null]
}

Table comisiones_plataforma {
  id bigint [pk, increment]
  pago_id bigint [not null, unique]
  importe_base decimal(10,2) [not null]
  porcentaje decimal(5,4) [not null]
  importe_comision decimal(10,2) [not null]
  ingreso_establecimiento decimal(10,2) [not null]
  pendiente_por_efectivo boolean [not null]
  fecha_calculo datetime [not null]
}

// =====================================================
// MODULO: EXPERIENCIA
// =====================================================

Table resenas {
  id bigint [pk, increment]
  reserva_id bigint [not null, unique]
  consumidor_id bigint [not null]
  establecimiento_id bigint [not null]
  calificacion int [not null, note: 'Valor entre 1 y 5']
  comentario varchar(1000)
  fecha_publicacion datetime [not null]
  visible boolean [not null]
}

Table notificaciones {
  id bigint [pk, increment]
  usuario_id bigint [not null]
  tipo varchar(35) [not null, note: 'RESULTADO_AFILIACION, DOCUMENTO_POR_VENCER, SUSPENSION, RECORDATORIO_RECOJO, CAMBIO_RESERVA o GENERAL']
  titulo varchar(120) [not null]
  mensaje varchar(1000) [not null]
  tipo_referencia varchar(60)
  referencia_id bigint [note: 'Referencia lógica; no posee clave foránea porque puede apuntar a distintos tipos de registro']
  fecha_creacion datetime [not null]
  leida boolean [not null]
  fecha_lectura datetime
}

Table impactos_co2_evitado {
  id bigint [pk, increment]
  recojo_id bigint [not null, unique]
  alimento_recuperado_kg decimal(10,3)
  co2_evitado_kg decimal(10,3)
  estimacion_disponible boolean [not null]
  metodologia varchar(300)
  fecha_registro datetime [not null]
}
